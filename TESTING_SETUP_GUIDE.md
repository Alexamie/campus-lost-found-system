# Complete NestJS Testing Environment Setup Guide

## Overview
This guide provides a complete setup for testing a NestJS project using Jest and Supertest, including installation, configuration, and sample tests.

---

## Part 1: Installation Commands

### Step 1: Install Jest and Dependencies
```bash
npm install --save-dev @nestjs/testing jest ts-jest @types/jest
npm install --save-dev supertest @types/supertest
npm install --save-dev jest-coverage-report ts-node
```

### Step 2: Verify Installation
```bash
npm ls jest
npm ls supertest
```

### Step 3: Create package.json Scripts
Add these scripts to your `package.json`:
```json
{
  "scripts": {
    "test": "jest",
    "test:watch": "jest --watch",
    "test:cov": "jest --coverage",
    "test:debug": "node --inspect-brk -r tsconfig-paths/register -r ts-node/register node_modules/.bin/jest --runInBand",
    "test:e2e": "jest --config ./test/jest-e2e.json"
  }
}
```

## Part 2: Configuration Files

### jest.config.js (Root Level - Unit & Integration Tests)
```javascript
module.exports = {
  moduleFileExtensions: ['js', 'json', 'ts'],
  rootDir: 'src',
  testRegex: '.*\\.spec\\.ts$',
  transform: {
    '^.+\\.(t|j)s$': 'ts-jest',
  },
  collectCoverageFrom: [
    '**/*.(t|j)s',
    '!**/*.spec.ts',
    '!**/index.ts',
  ],
  coverageDirectory: '../coverage',
  testEnvironment: 'node',
  roots: ['<rootDir>'],
  moduleNameMapper: {
    '^src/(.*)$': '<rootDir>/$1',
  },
};
```

## Part 4: Sample Tests

### 4.1 Unit Test Example: users.service.spec.ts

```typescript
import { Test, TestingModule } from '@nestjs/testing';
import { UsersService } from './users.service';
import { CreateUserDto } from './dto/create-user.dto';

describe('UsersService (Unit)', () => {
  let service: UsersService;
  let mockUsersRepository: any;

  beforeEach(async () => {
    // Mock the database repository
    mockUsersRepository = {
      create: jest.fn(),
      save: jest.fn(),
      find: jest.fn(),
      findOne: jest.fn(),
    };

    const module: TestingModule = await Test.createTestingModule({
      providers: [
        UsersService,
        {
          provide: 'USERS_REPOSITORY',
          useValue: mockUsersRepository,
        },
      ],
    }).compile();

    service = module.get<UsersService>(UsersService);
  });

  afterEach(() => {
    jest.clearAllMocks();
  });

  describe('create', () => {
    it('should create a new user', async () => {
      const createUserDto: CreateUserDto = {
        name: 'John Doe',
        email: 'john@example.com',
        password: 'password123',
      };

      const user = {
        id: 1,
        ...createUserDto,
        createdAt: new Date(),
      };

      mockUsersRepository.create.mockReturnValue(createUserDto);
      mockUsersRepository.save.mockResolvedValue(user);

      const result = await service.create(createUserDto);

      expect(result).toEqual(user);
      expect(mockUsersRepository.create).toHaveBeenCalledWith(createUserDto);
      expect(mockUsersRepository.save).toHaveBeenCalled();
    });
  });

  describe('findAll', () => {
    it('should return an array of users', async () => {
      const users = [
        { id: 1, name: 'John', email: 'john@example.com' },
        { id: 2, name: 'Jane', email: 'jane@example.com' },
      ];

      mockUsersRepository.find.mockResolvedValue(users);

      const result = await service.findAll();

      expect(result).toEqual(users);
      expect(mockUsersRepository.find).toHaveBeenCalled();
    });
  });

  describe('findOne', () => {
    it('should return a single user by id', async () => {
      const user = { id: 1, name: 'John', email: 'john@example.com' };

      mockUsersRepository.findOne.mockResolvedValue(user);

      const result = await service.findOne(1);

      expect(result).toEqual(user);
      expect(mockUsersRepository.findOne).toHaveBeenCalledWith({ where: { id: 1 } });
    });

    it('should return null if user not found', async () => {
      mockUsersRepository.findOne.mockResolvedValue(null);

      const result = await service.findOne(999);

      expect(result).toBeNull();
    });
  });
});
```

### 4.2 Integration Test Example: users-products.integration.spec.ts

```typescript
import { Test, TestingModule } from '@nestjs/testing';
import { UsersService } from '../users/users.service';
import { ProductsService } from '../products/products.service';

describe('Users and Products Integration', () => {
  let usersService: UsersService;
  let productsService: ProductsService;
  let mockUsersRepository: any;
  let mockProductsRepository: any;

  beforeEach(async () => {
    mockUsersRepository = {
      create: jest.fn(),
      save: jest.fn(),
      find: jest.fn(),
      findOne: jest.fn(),
    };

    mockProductsRepository = {
      create: jest.fn(),
      save: jest.fn(),
      find: jest.fn(),
      findOne: jest.fn(),
    };

    const module: TestingModule = await Test.createTestingModule({
      providers: [
        UsersService,
        ProductsService,
        {
          provide: 'USERS_REPOSITORY',
          useValue: mockUsersRepository,
        },
        {
          provide: 'PRODUCTS_REPOSITORY',
          useValue: mockProductsRepository,
        },
      ],
    }).compile();

    usersService = module.get<UsersService>(UsersService);
    productsService = module.get<ProductsService>(ProductsService);
  });

  describe('User purchases product', () => {
    it('should associate product with user', async () => {
      // Create user
      const user = {
        id: 1,
        name: 'John Doe',
        email: 'john@example.com',
      };

      // Create product
      const product = {
        id: 1,
        name: 'Laptop',
        price: 999.99,
        userId: user.id,
      };

      mockUsersRepository.findOne.mockResolvedValue(user);
      mockProductsRepository.create.mockReturnValue(product);
      mockProductsRepository.save.mockResolvedValue(product);

      // Purchase flow
      const foundUser = await usersService.findOne(user.id);
      const newProduct = await productsService.create({
        name: 'Laptop',
        price: 999.99,
        userId: foundUser.id,
      });

      expect(foundUser.id).toBe(user.id);
      expect(newProduct.userId).toBe(user.id);
    });
  });

  describe('Get all products for user', () => {
    it('should retrieve all products owned by user', async () => {
      const userId = 1;
      const userProducts = [
        { id: 1, name: 'Laptop', userId },
        { id: 2, name: 'Mouse', userId },
      ];

      mockProductsRepository.find.mockResolvedValue(userProducts);

      const result = await productsService.findByUserId(userId);

      expect(result).toHaveLength(2);
      expect(result.every(p => p.userId === userId)).toBe(true);
    });
  });
});
```

### 4.3 E2E Test Example: test/app.e2e-spec.ts

```typescript
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication, ValidationPipe } from '@nestjs/common';
import * as request from 'supertest';
import { AppModule } from './../src/app.module';

describe('App (e2e)', () => {
  let app: INestApplication;
  let userId: number;
  let productId: number;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();
    app.useGlobalPipes(new ValidationPipe());
    await app.init();
  });

  afterAll(async () => {
    await app.close();
  });

  describe('Users CRUD (POST, GET, PUT, DELETE)', () => {
    it('POST /users - should create a new user', () => {
      return request(app.getHttpServer())
        .post('/users')
        .send({
          name: 'John Doe',
          email: 'john@example.com',
          password: 'password123',
        })
        .expect(201)
        .expect((res) => {
          userId = res.body.id;
          expect(res.body.name).toBe('John Doe');
          expect(res.body.email).toBe('john@example.com');
          expect(res.body).toHaveProperty('id');
        });
    });

    it('GET /users - should retrieve all users', () => {
      return request(app.getHttpServer())
        .get('/users')
        .expect(200)
        .expect((res) => {
          expect(Array.isArray(res.body)).toBe(true);
          expect(res.body.length).toBeGreaterThan(0);
        });
    });

    it('GET /users/:id - should retrieve user by id', () => {
      return request(app.getHttpServer())
        .get(`/users/${userId}`)
        .expect(200)
        .expect((res) => {
          expect(res.body.id).toBe(userId);
          expect(res.body.name).toBe('John Doe');
        });
    });

    it('PUT /users/:id - should update user', () => {
      return request(app.getHttpServer())
        .put(`/users/${userId}`)
        .send({ name: 'Jane Doe' })
        .expect(200)
        .expect((res) => {
          expect(res.body.name).toBe('Jane Doe');
        });
    });

    it('DELETE /users/:id - should delete user', () => {
      return request(app.getHttpServer())
        .delete(`/users/${userId}`)
        .expect(200);
    });
  });

  describe('Products CRUD', () => {
    it('POST /products - should create a product', () => {
      return request(app.getHttpServer())
        .post('/products')
        .send({
          name: 'Laptop',
          description: 'High-performance laptop',
          price: 1299.99,
        })
        .expect(201)
        .expect((res) => {
          productId = res.body.id;
          expect(res.body.name).toBe('Laptop');
          expect(res.body.price).toBe(1299.99);
        });
    });

    it('GET /products - should retrieve all products', () => {
      return request(app.getHttpServer())
        .get('/products')
        .expect(200)
        .expect((res) => {
          expect(Array.isArray(res.body)).toBe(true);
        });
    });

    it('GET /products/:id - should retrieve product by id', () => {
      return request(app.getHttpServer())
        .get(`/products/${productId}`)
        .expect(200)
        .expect((res) => {
          expect(res.body.id).toBe(productId);
        });
    });
  });

  describe('Error Handling', () => {
    it('GET /users/999 - should return 404 for non-existent user', () => {
      return request(app.getHttpServer())
        .get('/users/999')
        .expect(404);
    });

    it('POST /users - should return 400 for invalid data', () => {
      return request(app.getHttpServer())
        .post('/users')
        .send({
          name: 'John',
          // Missing email and password
        })
        .expect(400);
    });
  });
});
```

## Part 7: Best Practices

### 1. Use Descriptive Test Names
```typescript
// ✅ Good
it('should create user with valid email and password', () => {});

// ❌ Bad
it('creates user', () => {});
```

### 2. Follow AAA Pattern (Arrange, Act, Assert)
```typescript
it('should calculate total price with tax', () => {
  // Arrange
  const price = 100;
  const taxRate = 0.1;

  // Act
  const result = calculateTax(price, taxRate);

  // Assert
  expect(result).toBe(110);
});
```

### 3. Mock External Dependencies
```typescript
const mockDatabase = {
  save: jest.fn(),
};

// Use mockDatabase instead of real DB
```

### 4. Test Edge Cases
```typescript
it('should handle null input gracefully', () => {
  expect(() => service.process(null)).not.toThrow();
});
```

### 5. Use BeforeEach/AfterEach for Setup/Cleanup
```typescript
beforeEach(async () => {
  // Initialize before each test
});

afterEach(() => {
  // Clean up after each test
  jest.clearAllMocks();
});
```

## Part 8: Troubleshooting Common Issues

### Issue: "Cannot find module" errors
**Solution**: Check `tsconfig.json` paths configuration and `rootDir` in jest.config.js

### Issue: Tests timeout
**Solution**: Increase timeout in jest configuration
```javascript
testTimeout: 30000, // 30 seconds
```

### Issue: Async errors not caught
**Solution**: Always return promises or use async/await
```typescript
// ✅ Correct
it('should work', async () => {
  const result = await service.method();
  expect(result).toBeDefined();
});

// ❌ Wrong
it('should work', () => {
  service.method(); // Promise not awaited
});
```
